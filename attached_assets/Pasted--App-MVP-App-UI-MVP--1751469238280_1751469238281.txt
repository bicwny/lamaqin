### **⚙️ 关键配置要点**

#### **1. Supabase客户端配置**
```typescript
// lib/supabase.ts
import { createClient } from '@supabase/supabase-js';
import { Database } from '@/types/database';

const supabaseUrl = process.env.EXPO_PUBLIC_SUPABASE_URL!;
const supabaseAnonKey = process.env.EXPO_PUBLIC_SUPABASE_ANON_KEY!;

export const supabase = createClient<Database>(supabaseUrl, supabaseAnonKey, {
  auth: {
    autoRefreshToken: true,
    persistSession: true,
    detectSessionInUrl: false,
  },
});
```

#### **2. 认证上下文设置**
```typescript
// components/auth/AuthProvider.tsx
import React, { createContext, useContext, useEffect, useState } from 'react';
import { Session, User } from '@supabase/supabase-js';
import { supabase } from '@/lib/supabase';

interface AuthContextType {
  user: User | null;
  session: Session | null;
  loading: boolean;
  signOut: () => Promise<void>;
}

const AuthContext = createContext<AuthContextType>({} as AuthContextType);

export function AuthProvider({ children }: { children: React.ReactNode }) {
  const [user, setUser] = useState<User | null>(null);
  const [session, setSession] = useState<Session | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    // 获取初始会话
    supabase.auth.getSession().then(({ data: { session } }) => {
      setSession(session);
      setUser(session?.user ?? null);
      setLoading(false);
    });

    // 监听认证状态变化
    const { data: { subscription } } = supabase.auth.onAuthStateChange(
      async (event, session) => {
        setSession(session);
        setUser(session?.user ?? null);
        setLoading(false);
      }
    );

    return () => subscription.unsubscribe();
  }, []);

  const signOut = async () => {
    await supabase.auth.signOut();
  };

  return (
    <AuthContext.Provider value={{ user, session, loading, signOut }}>
      {children}
    </AuthContext.Provider>
  );
}

export const useAuth = () => {
  const context = useContext(AuthContext);
  if (!context) {
    throw new Error('useAuth must be used within an AuthProvider');
  }
  return context;
};
```

## 📋 产品简介

### **应用概述**
佛教修行追踪App是一个综合性移动应用，支持修行者记录和追踪完整的修行活动，涵盖戒定慧三学的全面实践。

### **核心功能模块**
1. **闻思学习** - 课程听讲记录和进度管理
2. **修行记录** - 修法计数、观修时间、主题管理
3. **心性观照** - 善恶心记录和趋势分析  
4. **统计分析** - 全面的修行数据统计和可视化
5. **个人中心** - 用户管理、设置和数据导出

### **主界面导航结构**
```
📚 闻思  |  📿 修行  |  💝 心性  |  📊 统计  |  👤 个人
```

## 🗄️ 数据模型

### **核心表结构 (12张表)**

#### **用户管理**
```sql
-- 用户信息表
users (
  id UUID PRIMARY KEY,
  dharma_name VARCHAR(100),           -- 法名
  email VARCHAR(255) UNIQUE,          -- 邮箱
  location VARCHAR(100),              -- 常住地
  practice_years INTEGER,             -- 修行年限
  class_name VARCHAR(100),            -- 班级
  main_practices JSONB,               -- 修行偏好
  notification_settings JSONB,        -- 通知设置
  created_at TIMESTAMP,
  updated_at TIMESTAMP
)
```

#### **修行体系**
```sql
-- 修行主题表
themes (
  id UUID PRIMARY KEY,
  name VARCHAR(100) NOT NULL,         -- 主题名称：五加行、金刚萨埵法会
  description TEXT,                   -- 主题描述
  type VARCHAR(50) NOT NULL,          -- 类型：foundation, puja, meditation
  created_at TIMESTAMP
)

-- 修法表
practices (
  id UUID PRIMARY KEY,
  name VARCHAR(100) NOT NULL,         -- 修法名称：顶礼、百字明
  type VARCHAR(50) NOT NULL,          -- 类型：count(计数) 或 time(时间)
  unit VARCHAR(20) NOT NULL,          -- 单位：次 或 分钟
  description TEXT,
  created_at TIMESTAMP
)

-- 主题修法关系表
theme_practices (
  id UUID PRIMARY KEY,
  theme_id UUID REFERENCES themes(id),
  practice_id UUID REFERENCES practices(id),
  target_count INTEGER NOT NULL,      -- 目标数量：100000
  is_required BOOLEAN DEFAULT true,
  created_at TIMESTAMP
)
```

#### **课程体系**
```sql
-- 课程表
courses (
  id UUID PRIMARY KEY,
  name VARCHAR(100) NOT NULL,         -- 课程名称：《入菩萨行论》
  total_lessons INTEGER NOT NULL,     -- 总课数：201
  teacher VARCHAR(100),               -- 授课上师
  description TEXT,
  created_at TIMESTAMP
)

-- 课程章节表
course_lessons (
  id UUID PRIMARY KEY,
  course_id UUID REFERENCES courses(id),
  lesson_number INTEGER NOT NULL,     -- 课次：1, 2, 3...
  title VARCHAR(200) NOT NULL,        -- 课程标题
  content_summary TEXT,               -- 内容摘要
  created_at TIMESTAMP
)
```

#### **用户活动数据**
```sql
-- 用户修行项目表
user_practice_projects (
  id UUID PRIMARY KEY,
  user_id UUID REFERENCES users(id),
  theme_id UUID REFERENCES themes(id),
  practice_id UUID REFERENCES practices(id),
  target_count INTEGER NOT NULL,      -- 目标数量
  start_date DATE,                    -- 开始日期
  target_end_date DATE,               -- 目标完成日期
  current_count INTEGER DEFAULT 0,    -- 当前进度
  daily_target INTEGER DEFAULT 0,     -- 每日目标
  status VARCHAR(20) DEFAULT 'not_started', -- not_started, active, completed
  created_at TIMESTAMP,
  updated_at TIMESTAMP
)

-- 每日修行记录表
daily_records (
  id UUID PRIMARY KEY,
  user_id UUID REFERENCES users(id),
  practice_project_id UUID REFERENCES user_practice_projects(id),
  record_date DATE NOT NULL,          -- 记录日期
  count INTEGER NOT NULL,             -- 完成数量
  notes TEXT,                         -- 备注
  created_at TIMESTAMP
)

-- 观修记录表
meditation_records (
  id UUID PRIMARY KEY,
  user_id UUID REFERENCES users(id),
  practice_id UUID REFERENCES practices(id),
  record_date DATE NOT NULL,
  session_number INTEGER NOT NULL,    -- 第几座 (1-92)
  duration_minutes INTEGER NOT NULL,  -- 观修时长(分钟)
  session_attempt INTEGER DEFAULT 1,  -- 这座的第几次 (1-3+)
  method VARCHAR(100),                -- 观修方法
  created_at TIMESTAMP
)

-- 听课记录表
study_records (
  id UUID PRIMARY KEY,
  user_id UUID REFERENCES users(id),
  course_id UUID REFERENCES courses(id),
  lesson_id UUID REFERENCES course_lessons(id),
  study_date DATE NOT NULL,
  study_count_for_lesson INTEGER DEFAULT 1, -- 该课的第几次学习
  created_at TIMESTAMP
)

-- 观心记录表
mindfulness_records (
  id UUID PRIMARY KEY,
  user_id UUID REFERENCES users(id),
  record_date DATE NOT NULL,
  record_time TIME NOT NULL,
  mind_type VARCHAR(10) NOT NULL,     -- 'good'(善心) 或 'bad'(恶心)
  description TEXT,                   -- 具体描述
  created_at TIMESTAMP
)

-- 一次性修行记录表
one_time_records (
  id UUID PRIMARY KEY,
  user_id UUID REFERENCES users(id),
  record_date DATE NOT NULL,
  practice_type VARCHAR(50) NOT NULL, -- 计数修法、观修记录、功课诵读、持戒记录、其他修行
  practice_name VARCHAR(100) NOT NULL,
  count_or_duration VARCHAR(50) NOT NULL,
  notes TEXT,
  created_at TIMESTAMP
)

-- 持戒记录表
precept_records (
  id UUID PRIMARY KEY,
  user_id UUID REFERENCES users(id),
  precept_type VARCHAR(50) NOT NULL,  -- 八关斋戒、菩萨戒、诵戒、荟供、密乘戒
  record_date DATE NOT NULL,
  notes TEXT,
  created_at TIMESTAMP
)
```

```typescript
interface MeditationSession {
  sessionNumber: number;      // 1-92
  durationMinutes: number;   // 本次时长
  sessionAttempt: number;    // 该座的第几次
}

// 验证规则
function validateMeditationSession(sessions: MeditationSession[]) {
  const totalDuration = sessions.reduce((sum, s) => sum + s.durationMinutes, 0);
  const validSessions = sessions.filter(s => s.durationMinutes >= 15);
  
  // 规则: 总时长≥30分钟 或 每座≥15分钟且有2座以上
  return totalDuration >= 30 || (validSessions.length >= 2);
}
```
