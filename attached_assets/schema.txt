CREATE TABLE public.course_lessons (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  course_id uuid,
  lesson_number integer NOT NULL,
  title character varying NOT NULL,
  content_summary text,
  created_at timestamp with time zone DEFAULT now(),
  CONSTRAINT course_lessons_pkey PRIMARY KEY (id),
  CONSTRAINT course_lessons_course_id_fkey FOREIGN KEY (course_id) REFERENCES public.courses(id)
);
CREATE TABLE public.courses (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  name character varying NOT NULL,
  total_lessons integer NOT NULL,
  teacher character varying,
  description text,
  created_at timestamp with time zone DEFAULT now(),
  CONSTRAINT courses_pkey PRIMARY KEY (id)
);
CREATE TABLE public.daily_records (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  user_id uuid,
  practice_project_id uuid,
  record_date date NOT NULL,
  count integer NOT NULL,
  notes text,
  created_at timestamp with time zone DEFAULT now(),
  CONSTRAINT daily_records_pkey PRIMARY KEY (id),
  CONSTRAINT daily_records_practice_project_id_fkey FOREIGN KEY (practice_project_id) REFERENCES public.user_practice_projects(id),
  CONSTRAINT daily_records_user_id_fkey FOREIGN KEY (user_id) REFERENCES public.users(id)
);
CREATE TABLE public.meditation_records (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  user_id uuid,
  practice_id uuid,
  record_date date NOT NULL,
  session_number integer NOT NULL,
  duration_minutes integer NOT NULL,
  session_attempt integer DEFAULT 1,
  method character varying,
  created_at timestamp with time zone DEFAULT now(),
  CONSTRAINT meditation_records_pkey PRIMARY KEY (id),
  CONSTRAINT meditation_records_user_id_fkey FOREIGN KEY (user_id) REFERENCES public.users(id),
  CONSTRAINT meditation_records_practice_id_fkey FOREIGN KEY (practice_id) REFERENCES public.practices(id)
);
CREATE TABLE public.mindfulness_records (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  user_id uuid,
  record_date date NOT NULL,
  record_time time without time zone NOT NULL,
  mind_type character varying NOT NULL CHECK (mind_type::text = ANY (ARRAY['good'::character varying, 'bad'::character varying]::text[])),
  description text,
  created_at timestamp with time zone DEFAULT now(),
  CONSTRAINT mindfulness_records_pkey PRIMARY KEY (id),
  CONSTRAINT mindfulness_records_user_id_fkey FOREIGN KEY (user_id) REFERENCES public.users(id)
);
CREATE TABLE public.one_time_records (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  user_id uuid,
  record_date date NOT NULL,
  practice_type character varying NOT NULL,
  practice_name character varying NOT NULL,
  count_or_duration character varying NOT NULL,
  notes text,
  created_at timestamp with time zone DEFAULT now(),
  CONSTRAINT one_time_records_pkey PRIMARY KEY (id),
  CONSTRAINT one_time_records_user_id_fkey FOREIGN KEY (user_id) REFERENCES public.users(id)
);
CREATE TABLE public.practices (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  name character varying NOT NULL,
  type character varying NOT NULL CHECK (type::text = ANY (ARRAY['count'::character varying, 'time'::character varying]::text[])),
  unit character varying NOT NULL,
  description text,
  created_at timestamp with time zone DEFAULT now(),
  CONSTRAINT practices_pkey PRIMARY KEY (id)
);
CREATE TABLE public.precept_records (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  user_id uuid,
  precept_type character varying NOT NULL,
  record_date date NOT NULL,
  notes text,
  created_at timestamp with time zone DEFAULT now(),
  CONSTRAINT precept_records_pkey PRIMARY KEY (id),
  CONSTRAINT precept_records_user_id_fkey FOREIGN KEY (user_id) REFERENCES public.users(id)
);
CREATE TABLE public.study_records (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  user_id uuid,
  course_id uuid,
  lesson_id uuid,
  study_date date NOT NULL,
  study_count_for_lesson integer DEFAULT 1,
  created_at timestamp with time zone DEFAULT now(),
  CONSTRAINT study_records_pkey PRIMARY KEY (id),
  CONSTRAINT study_records_user_id_fkey FOREIGN KEY (user_id) REFERENCES public.users(id),
  CONSTRAINT study_records_lesson_id_fkey FOREIGN KEY (lesson_id) REFERENCES public.course_lessons(id),
  CONSTRAINT study_records_course_id_fkey FOREIGN KEY (course_id) REFERENCES public.courses(id)
);
CREATE TABLE public.theme_practices (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  theme_id uuid,
  practice_id uuid,
  target_count integer NOT NULL,
  is_required boolean DEFAULT true,
  created_at timestamp with time zone DEFAULT now(),
  CONSTRAINT theme_practices_pkey PRIMARY KEY (id),
  CONSTRAINT theme_practices_theme_id_fkey FOREIGN KEY (theme_id) REFERENCES public.themes(id),
  CONSTRAINT theme_practices_practice_id_fkey FOREIGN KEY (practice_id) REFERENCES public.practices(id)
);
CREATE TABLE public.themes (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  name character varying NOT NULL,
  description text,
  type character varying NOT NULL,
  created_at timestamp with time zone DEFAULT now(),
  CONSTRAINT themes_pkey PRIMARY KEY (id)
);
CREATE TABLE public.user_practice_projects (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  user_id uuid,
  theme_id uuid,
  practice_id uuid,
  target_count integer NOT NULL,
  start_date date,
  target_end_date date,
  current_count integer DEFAULT 0,
  daily_target integer DEFAULT 0,
  status character varying DEFAULT 'not_started'::character varying CHECK (status::text = ANY (ARRAY['not_started'::character varying, 'active'::character varying, 'completed'::character varying]::text[])),
  created_at timestamp with time zone DEFAULT now(),
  updated_at timestamp with time zone DEFAULT now(),
  CONSTRAINT user_practice_projects_pkey PRIMARY KEY (id),
  CONSTRAINT user_practice_projects_theme_id_fkey FOREIGN KEY (theme_id) REFERENCES public.themes(id),
  CONSTRAINT user_practice_projects_user_id_fkey FOREIGN KEY (user_id) REFERENCES public.users(id),
  CONSTRAINT user_practice_projects_practice_id_fkey FOREIGN KEY (practice_id) REFERENCES public.practices(id)
);
CREATE TABLE public.users (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  dharma_name character varying,
  email character varying UNIQUE,
  location character varying,
  practice_years integer,
  class_name character varying,
  created_at timestamp with time zone DEFAULT now(),
  updated_at timestamp with time zone DEFAULT now(),
  CONSTRAINT users_pkey PRIMARY KEY (id)
);
