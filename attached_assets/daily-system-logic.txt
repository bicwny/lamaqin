
# DAILY SYSTEM LOGIC - Buddhist Practice App

## OVERVIEW
The daily system serves as the central hub for the Buddhist practice app, providing users with a unified dashboard that aggregates all daily activities - study, practice, and mindfulness tracking. It implements timezone-aware daily resets and provides quick actions for maintaining consistent spiritual practice.

## 2. TIMEZONE-AWARE DAILY RESET SYSTEM

### 2.1 Timezone Detection & Management
```typescript
// From hooks/useTimezone.ts
const TimezoneSystem = {
  initialization: async () => {
    // Detect user's local timezone
    const timezone = Intl.DateTimeFormat().resolvedOptions().timeZone;
    const offset = new Date().getTimezoneOffset();
    
    return {
      timezone,
      offset,
      displayName: `${timezone} (GMT${offset >= 0 ? '-' : '+'}${Math.abs(offset/60)})`
    };
  },
  
  getCurrentDateInTimezone: (timezone: string) => {
    // Return YYYY-MM-DD format in user's timezone
    const now = new Date();
    const localTime = new Date(now.toLocaleString("en-US", {timeZone: timezone}));
    return localTime.toISOString().split('T')[0];
  },
  
  dailyResetCheck: (timezoneInfo, lastResetDate, onReset) => {
    const currentDate = getCurrentDateInTimezone(timezoneInfo.timezone);
    
    if (lastResetDate && currentDate !== lastResetDate) {
      console.log('🌅 New day detected! Resetting daily counters...');
      onReset(); // Trigger UI reset
      AsyncStorage.setItem('lastResetDate', currentDate);
    }
  }
};
```

## 3. STUDY INTEGRATION IN DAILY DASHBOARD

### 3.1 Active Course Display Logic
```typescript
const StudyDashboardIntegration = {
  loadCourseLessons: async () => {
    // Get user's active courses only
    const { data: userCourses } = await supabase
      .from('user_courses')
      .select('*, course:courses(*)')
      .eq('user_id', user.id)
      .eq('status', 'active')
      .order('progress_percentage', { ascending: false });
    
    const courseCards = [];
    
    for (const userCourse of userCourses) {
      // Find next incomplete lesson
      const nextLesson = await findNextIncompleteLesson(userCourse);
      
      if (nextLesson) {
        courseCards.push({
          courseId: userCourse.course_id,
          courseName: userCourse.course.name,
          lessonNumber: nextLesson.lesson_number,
          lessonTitle: nextLesson.title,
          progress: `听传承: ${listenCount}次 | 看法本: ${readCount}次`,
          teacher: userCourse.course.teacher
        });
      }
    }
    
    return courseCards;
  },
  
  findNextIncompleteLesson: async (userCourse) => {
    // Complex logic to determine next lesson needing completion
    // A lesson is complete only when BOTH 听传承 AND 看法本 are recorded
    
    const studyRecords = await getStudyRecords(userCourse.course_id);
    const allLessons = await getCourseLessons(userCourse.course_id);
    
    // Group records by lesson and check completion
    const lessonCompletionMap = buildCompletionMap(studyRecords);
    
    // Find first lesson that doesn't have both study types
    for (let i = 1; i <= userCourse.course.total_lessons; i++) {
      const lessonData = lessonCompletionMap.get(i);
      if (!lessonData || !lessonData.听传承 || !lessonData.看法本) {
        return allLessons.find(l => l.lesson_number === i);
      }
    }
    
    return null; // All lessons completed
  }
};
```

const StudyQuickActions = {

      loadCourseLessons();
      toastService.error({
        title: '保存失败',
        message: '网络异常，请稍后重试'
      });
    }
  }

### 4.2 Practice Loading Logic
```typescript
const PracticeDashboardIntegration = {
  loadDailyPractices: async () => {
    // Load count-based practices with daily targets
    const { data: projects } = await supabase
      .from('user_practice_projects')
      .select('*, practices(*)')
      .eq('user_id', user.id)
      .eq('status', 'active')
      .eq('target_period', 'daily')
      .limit(3);
    
    const today = getCurrentDateInTimezone(timezoneInfo.timezone);
    const practicesData = [];
    
    for (const project of projects) {
      if (project.practices.type === 'count') {
        // Get today's count using timezone-aware date
        const { data: todayRecords } = await supabase
          .from('daily_records')
          .select('count')
          .eq('user_id', user.id)
          .eq('practice_project_id', project.id)
          .eq('record_date', today);
        
        const todayCount = todayRecords?.reduce((sum, r) => sum + r.count, 0) || 0;
        const progressPercent = Math.min((todayCount / project.daily_target) * 100, 100);
        
        let status = 'pending';
        if (todayCount >= project.daily_target) status = 'completed';
        else if (todayCount > 0) status = 'in_progress';
        
        practicesData.push({
          id: project.id,
          name: project.practices.name,
          current: todayCount,
          target: project.daily_target,
          unit: project.practices.unit,
          status,
          progressPercent,
          type: 'count'
        });
      }
    }
    
    setDailyPractices(practicesData);
  },
  
  loadWeeklyPractices: async () => {
    // Load time-based practices with weekly targets
    const { data: projects } = await supabase
      .from('user_practice_projects')
      .select('*, practices(*)')
      .eq('user_id', user.id)
      .eq('status', 'active')
      .eq('target_period', 'weekly');
    
    const today = getCurrentDateInTimezone(timezoneInfo.timezone);
    const weekStart = getWeekStart(today); // Monday-based week
    
    const practicesData = [];
    
    for (const project of projects) {
      if (project.practices.type === 'time') {
        // Get this week's meditation sessions
        const { data: weekRecords } = await supabase
          .from('meditation_records')
          .select('duration_minutes, record_date')
          .eq('user_id', user.id)
          .eq('practice_id', project.practice_id)
          .gte('record_date', weekStart)
          .lte('record_date', today);
        
        const weekSessions = weekRecords?.length || 0;
        const todayRecords = weekRecords?.filter(r => r.record_date === today) || [];
        const todaySessions = todayRecords.length;
        
        let status = 'pending';
        if (weekSessions >= project.daily_target) status = 'completed';
        else if (weekSessions > 0) status = 'in_progress';
        
        const todayDetails = todayRecords.length > 0 
          ? todayRecords.map((r, i) => `第${i + 1}座${r.duration_minutes}分钟`).join('；')
          : undefined;
        
        practicesData.push({
          id: project.id,
          name: project.practices.name,
          weekSessions,
          weekTarget: project.daily_target,
          todaySessions,
          status,
          todayDetails,
          practiceId: project.practice_id,
          type: 'time'
        });
      }
    }
    
    setWeeklyPractices(practicesData);
  }
};
```

### 4.3 Quick Action System
```typescript
const PracticeQuickActions = {
  // Quick complete button (✓) - completes daily target immediately
  handleQuickComplete: async (practice) => {
    if (practice.status === 'completed') {
      toastService.info({ 
        title: '今日目标已达成', 
        message: '继续保持！明日再接再厉' 
      });
      return;
    }
    
    const remaining = practice.target - practice.current;
    
    // Optimistic UI update
    updatePracticeOptimistically(practice.id, remaining, practice.type);
    
    // Immediate success feedback
    toastService.success({
      title: '已完成今日目标',
      message: `${practice.name} +${remaining.toLocaleString()} ${practice.unit}`
    });
    
    // Background recording
    await recordQuickCompleteBackground(practice, remaining);
  },
  
  // Add record button (+) - navigates to detailed recording
  handleAddRecord: (practice) => {
    setLastRecordTime(Date.now()); // Prevent unnecessary refresh
    
    router.push({
      pathname: '/practice-detail/[practiceId]',
      params: { practiceId: practice.id }
    });
  },
  
  // Optimistic update for immediate UI feedback
  updatePracticeOptimistically: (practiceId, amount, practiceType) => {
    if (practiceType === 'time') {
      // Update weekly practices
      setWeeklyPractices(prev => prev.map(practice => {
        if (practice.id === practiceId) {
          const newWeekSessions = practice.weekSessions + 1;
          const newTodaySessions = practice.todaySessions + 1;
          
          return {
            ...practice,
            weekSessions: newWeekSessions,
            todaySessions: newTodaySessions,
            status: newWeekSessions >= practice.weekTarget ? 'completed' : 'in_progress',
            todayDetails: practice.todayDetails 
              ? `${practice.todayDetails}；第${newTodaySessions}座${amount}分钟`
              : `第${newTodaySessions}座${amount}分钟`
          };
        }
        return practice;
      }));
    } else {
      // Update daily practices
      setDailyPractices(prev => prev.map(practice => {
        if (practice.id === practiceId) {
          const newCurrent = practice.current + amount;
          const newProgressPercent = Math.min((newCurrent / practice.target) * 100, 100);
          
          return {
            ...practice,
            current: newCurrent,
            progressPercent: newProgressPercent,
            status: newCurrent >= practice.target ? 'completed' : 'in_progress'
          };
        }
        return practice;
      }));
    }
  }
};
```

const MindfulnessDashboardIntegration = {
  loadTodayMindfulness: async () => {
    const today = getCurrentDateInTimezone(timezoneInfo.timezone);

      .from('mindfulness_records')
      .select('mind_type')
      .eq('user_id', user.id)

      goodCount,
      badCount,
      total,
      mindfulnessRatio: `${mindfulnessRatio}%`,
      status: total > 0 ? 'active' : 'pending'
