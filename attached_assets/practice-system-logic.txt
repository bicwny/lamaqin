
# Practice System Logic & Implementation Guide
# React Native Expo App - 修行追踪 (Practice Tracking)

## Overview
This document outlines the complete practice tracking system architecture, covering all types of practice activities including count-based practices, time-based practices, meditation, study courses, and mindfulness tracking.

## Core Practice Architecture

### 1. Practice Types Classification
The system supports multiple practice tracking methods:

#### A. COUNT-BASED PRACTICES (计数类)
- **Unit**: 次 (times/repetitions)
- **Examples**: 佛号念诵, 礼拜, 大礼拜, 供灯, 供花
- **Tracking Method**: Increment counters for each completion
- **Progress Calculation**: current_count / target_count * 100%

#### B. TIME-BASED PRACTICES (计时类)
- **Unit**: 分钟 (minutes) or 座 (sessions)
- **Examples**: 观修, 禅修, 念佛
- **Tracking Method**: Duration recording with start/stop functionality
- **Progress Calculation**: total_minutes / target_minutes * 100%

#### C. MEDITATION PRACTICES (观修记录)
- **Unit**: 座 (meditation sessions)
- **Special Features**: 
  - 92 specific meditation topics (前行实修法)
  - Reflection recording (观后感)
  - Session numbering
  - Method tracking
- **Progress Calculation**: completed_sessions / 92 * 100%

#### D. STUDY PRACTICES (闻思学习)
- **Unit**: 课 (lessons)
- **Structure**: Courses → Lessons → Study Records
- **Study Types**: 听传承 (listening to teachings), 看法本 (reading texts)
- **Progress Calculation**: completed_lessons / total_lessons * 100%

#### E. MINDFULNESS TRACKING (觉察记录)
- **Unit**: 次 (instances)
- **Categories**: Good mind states vs Bad mind states
- **Tracking Method**: Real-time logging with timestamps
- **Analytics**: Daily/weekly trends, percentage calculations

const calculateCourseProgress = async (userId: string, courseId: string) => {
  // Get total lessons
  const course = await getCourse(courseId);
  
  // Get completed lessons (both 听传承 and 看法本)
  const studyRecords = await getStudyRecords(userId, courseId);
  
  // Group by lesson and check completion
  const completedLessons = groupByLesson(studyRecords)
    .filter(lesson => lesson.听传承 && lesson.看法本);
  
  const progress = (completedLessons.length / course.total_lessons) * 100;
  
  // Update user_courses table
  await updateCourseProgress(userId, courseId, progress);
  
  return progress;
};

#### Count-Based Practice Workflow
```typescript
const recordCountPractice = async (
  projectId: string,
  count: number,
  notes?: string
) => {
  const today = new Date().toISOString().split('T')[0];
  
  // Create daily record
  const dailyRecord = await supabase
    .from('daily_records')
    .insert({
      user_id: userId,
      practice_project_id: projectId,
      record_date: today,
      count: count,
      notes: notes || null
    });
  
  // Update project current count
  const { data: project } = await supabase
    .from('user_practice_projects')
    .select('current_count, target_count')
    .eq('id', projectId)
    .single();
  
  const newCurrentCount = project.current_count + count;
  const isCompleted = newCurrentCount >= project.target_count;
  
  await supabase
    .from('user_practice_projects')
    .update({
      current_count: newCurrentCount,
      status: isCompleted ? 'completed' : 'active',
      updated_at: new Date().toISOString()
    })
    .eq('id', projectId);
  
  return { success: true, isCompleted };
};
```

const createPracticeProject = async (config: ProjectConfiguration) => {
  let finalTargetCount: number;
  let finalDailyTarget: number;
  let endDate: Date | null = null;
  
  if (config.practiceType === 'count') {
    const days = calculateDurationDays(config);
    endDate = new Date(config.startDate.getTime() + days * 24 * 60 * 60 * 1000);
    
    if (config.configMode === 'total') {
      finalTargetCount = config.totalTarget!;
      finalDailyTarget = Math.ceil(finalTargetCount / days);
    } else {
      finalDailyTarget = config.dailyTarget!;
      finalTargetCount = finalDailyTarget * days;
    }
  } else {
    // Time-based practices
    finalDailyTarget = config.sessionsTarget;
    
    if (config.durationMode === '持续进行') {
      endDate = null;
      finalTargetCount = 0; // Indicates ongoing
    } else {
      const days = calculateDurationDays(config);
      endDate = new Date(config.startDate.getTime() + days * 24 * 60 * 60 * 1000);
      finalTargetCount = config.targetPeriod === 'weekly' 
        ? finalDailyTarget * Math.ceil(days / 7)
        : finalDailyTarget * days;
    }
  }
  
  const projectData = {
    user_id: userId,
    practice_id: config.practiceId,
    target_count: finalTargetCount,
    daily_target: finalDailyTarget,
    target_period: config.targetPeriod || 'daily',
    start_date: config.startDate.toISOString().split('T')[0],
    target_end_date: endDate ? endDate.toISOString().split('T')[0] : null,
    status: 'active',
    current_count: 0,
    goal_type: config.goalType,
    preset_project_id: config.presetProjectId || null,
    project_name: config.presetProjectId ? null : config.projectName || null,
  };
  
  const { data, error } = await supabase
    .from('user_practice_projects')
    .insert([projectData])

    const { data, error } = await supabase
      .from('meditation_records')
      .select('*')
      .eq('practice_id', practiceId)
      .eq('user_id', userId)
      .order('record_date', { ascending: false })
      .order('created_at', { ascending: false })
      .limit(limit);
    
    return data || [];
  },
  
  getProgressStatistics: async (projectId: string) => {
    const records = await this.getDailyRecords(projectId, 100);
    
    if (records.length === 0) return null;
    
    const totalCount = records.reduce((sum, r) => sum + r.count, 0);
    const averageDaily = totalCount / records.length;
    const activeDays = records.length;
    const streak = calculateCurrentStreak(records);
    
    return {
      totalCount,
      averageDaily: Math.round(averageDaily),
      activeDays,
      streak,
      firstRecord: records[records.length - 1].record_date,
      lastRecord: records[0].record_date
    };
  }
};

const calculateCurrentStreak = (records: DailyRecord[]) => {
  if (records.length === 0) return 0;
  
  // Sort by date descending
  const sortedRecords = records.sort((a, b) => 
    new Date(b.record_date).getTime() - new Date(a.record_date).getTime()
  );
  
  let streak = 0;
  let currentDate = new Date();
  
  for (const record of sortedRecords) {
    const recordDate = new Date(record.record_date);
    const daysDiff = Math.floor(
      (currentDate.getTime() - recordDate.getTime()) / (24 * 60 * 60 * 1000)
    );
    
    if (daysDiff === streak) {
      streak++;
      currentDate = recordDate;
    } else {
      break;
    }
  }
  
  return streak;
};

const PracticeValidation = {
  validateProjectData: (projectData: any) => {
    const errors = [];
    
    if (!projectData.user_id) errors.push('用户ID必填');
    if (!projectData.practice_id) errors.push('修行ID必填');
    if (!projectData.start_date) errors.push('开始日期必填');
    
    if (projectData.target_count <= 0) errors.push('目标数量必须大于0');
    if (projectData.daily_target <= 0) errors.push('每日目标必须大于0');
    
    if (projectData.target_end_date) {
      const startDate = new Date(projectData.start_date);
      const endDate = new Date(projectData.target_end_date);
      if (endDate <= startDate) {
        errors.push('结束日期必须晚于开始日期');
      }
    }
    
    if (errors.length > 0) {
      throw new Error(`项目数据验证失败: ${errors.join(', ')}`);
    }
    
    return true;
  },

  validateRecordData: (recordData: any, practiceType: string) => {
    const errors = [];
    
    if (!recordData.user_id) errors.push('用户ID必填');
    if (!recordData.record_date) errors.push('记录日期必填');
    
    if (practiceType === 'count') {
      if (!recordData.practice_project_id) errors.push('项目ID必填');
      if (!recordData.count || recordData.count <= 0) errors.push('数量必须大于0');
    } else {
      if (!recordData.practice_id) errors.push('修行ID必填');
      if (!recordData.duration_minutes || recordData.duration_minutes <= 0) {
        errors.push('时长必须大于0分钟');
      }
    }
    
    if (errors.length > 0) {
      throw new Error(`记录数据验证失败: ${errors.join(', ')}`);
    }
    
    return true;
  }
};

const DataConsistencyCheck = {
  auditProjectProgress: async () => {
    const { data: projects } = await supabase
      .from('user_practice_projects')
      .select('*');
    
    for (const project of projects || []) {
      if (project.practices?.type === 'count') {
        const { data: records } = await supabase
          .from('daily_records')
          .select('count')
          .eq('practice_project_id', project.id);
        
        const calculatedCount = records?.reduce((sum, r) => sum + r.count, 0) || 0;
        
        if (calculatedCount !== project.current_count) {
          console.warn('Count mismatch for project:', {
            projectId: project.id,
            stored: project.current_count,
            calculated: calculatedCount
          });
          
          // Auto-fix if needed
          await supabase
            .from('user_practice_projects')
            .update({ current_count: calculatedCount })
            .eq('id', project.id);
        }
      }
    }
  },

  cleanOrphanedRecords: async () => {
    // Check for daily records without valid projects
    const { data: orphanedDaily } = await supabase
      .from('daily_records')
      .select('id, practice_project_id')
      .is('user_practice_projects.id', null);
    
    if (orphanedDaily && orphanedDaily.length > 0) {
      console.warn('Found orphaned daily records:', orphanedDaily.length);
    }
    
    // Check for meditation records without valid practices
    const { data: orphanedMeditation } = await supabase
      .from('meditation_records')
      .select('id, practice_id')
      .is('practices.id', null);
    
    if (orphanedMeditation && orphanedMeditation.length > 0) {
      console.warn('Found orphaned meditation records:', orphanedMeditation.length);
    }
  }
