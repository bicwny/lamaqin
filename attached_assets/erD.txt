    %% 用户相关
    USERS {
        uuid id PK
        varchar dharma_name
        varchar email
        varchar location
        integer practice_years
        varchar class_name
        timestamp created_at
        timestamp updated_at
    }

    %% 修行体系
    THEMES {
        uuid id PK
        varchar name
        text description
        varchar type
        timestamp created_at
    }

    PRACTICES {
        uuid id PK
        varchar name
        varchar type
        varchar unit
        text description
        timestamp created_at
    }

    THEME_PRACTICES {
        uuid id PK
        uuid theme_id FK
        uuid practice_id FK
        integer target_count
        boolean is_required
        timestamp created_at
    }

    %% 课程体系
    COURSES {
        uuid id PK
        varchar name
        integer total_lessons
        varchar teacher
        text description
        timestamp created_at
    }

    COURSE_LESSONS {
        uuid id PK
        uuid course_id FK
        integer lesson_number
        varchar title
        text content_summary
        timestamp created_at
    }

    %% 用户修行项目
    USER_PRACTICE_PROJECTS {
        uuid id PK
        uuid user_id FK
        uuid theme_id FK
        uuid practice_id FK
        integer target_count
        date start_date
        date target_end_date
        integer current_count
        integer daily_target
        varchar status
        timestamp created_at
        timestamp updated_at
    }

    %% 记录系统
    DAILY_RECORDS {
        uuid id PK
        uuid user_id FK
        uuid practice_project_id FK
        date record_date
        integer count
        text notes
        timestamp created_at
    }

    MEDITATION_RECORDS {
        uuid id PK
        uuid user_id FK
        uuid practice_id FK
        date record_date
        integer session_number
        integer duration_minutes
        varchar method
        timestamp created_at
    }

    STUDY_RECORDS {
        uuid id PK
        uuid user_id FK
        uuid course_id FK
        uuid lesson_id FK
        date study_date
        integer study_count_for_lesson
        timestamp created_at
    }

    MINDFULNESS_RECORDS {
        uuid id PK
        uuid user_id FK
        date record_date
        time record_time
        varchar mind_type
        text description
        timestamp created_at
    }

    ONE_TIME_RECORDS {
        uuid id PK
        uuid user_id FK
        date record_date
        varchar practice_type
        varchar practice_name
        varchar count_or_duration
        text notes
        timestamp created_at
    }

    PRECEPT_RECORDS {
        uuid id PK
        uuid user_id FK
        varchar precept_type
        date record_date
        text notes
        timestamp created_at
    }

    %% 关系定义
    USERS ||--o{ USER_PRACTICE_PROJECTS : "has"
    USERS ||--o{ DAILY_RECORDS : "creates"
    USERS ||--o{ MEDITATION_RECORDS : "creates"
    USERS ||--o{ STUDY_RECORDS : "creates"
    USERS ||--o{ MINDFULNESS_RECORDS : "creates"
    USERS ||--o{ ONE_TIME_RECORDS : "creates"
    USERS ||--o{ PRECEPT_RECORDS : "creates"

    THEMES ||--o{ THEME_PRACTICES : "contains"
    THEMES ||--o{ USER_PRACTICE_PROJECTS : "belongs_to"
    
    PRACTICES ||--o{ THEME_PRACTICES : "included_in"
    PRACTICES ||--o{ USER_PRACTICE_PROJECTS : "associated_with"
    PRACTICES ||--o{ MEDITATION_RECORDS : "used_in"

    COURSES ||--o{ COURSE_LESSONS : "has"
    COURSES ||--o{ STUDY_RECORDS : "studied_in"
    COURSE_LESSONS ||--o{ STUDY_RECORDS : "lesson_studied"

    USER_PRACTICE_PROJECTS ||--o{ DAILY_RECORDS : "tracked_by"
